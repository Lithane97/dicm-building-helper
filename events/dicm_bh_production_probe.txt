# Console diagnostics only. No construction or spending.
# event dicm_bh_production_probe.1            -> capital province
# event dicm_bh_production_probe.2 PROVINCE_ID -> specified province
namespace = dicm_bh_production_probe
country_event = {
    id = dicm_bh_production_probe.1
    hidden = yes
    is_triggered_only = yes
    immediate = {
        capital_scope = { province_event = { id = dicm_bh_production_probe.2 } }
    }
    option = { name = dicm_bh_production_probe_close }
}
province_event = {
    id = dicm_bh_production_probe.2
    title = dicm_bh_production_probe_title
    desc = dicm_bh_production_probe_desc
    picture = TRADE_eventPicture
    is_triggered_only = yes
    immediate = {
        hidden_effect = {
            dicm_bh_calculate_production_gains = yes
            set_variable = { which = dicm_bh_production_minimum value = 0.1 }
            set_variable = { which = dicm_bh_production_selection value = 0 }
            set_variable = { which = dicm_bh_production_selected_gain value = 0 }
            clr_province_flag = dicm_bh_production_probe_selected
            clr_province_flag = dicm_bh_production_probe_pass
            owner = {
                set_variable = { which = dicm_bh_production_minimum which = dicm_bh_minimum }
                set_variable = { which = dicm_bh_production_selection which = dicm_bh_selection }
                ROOT = {
                    set_variable = { which = dicm_bh_production_minimum which = PREV }
                    set_variable = { which = dicm_bh_production_selection which = PREV }
                }
            }
            if = {
                limit = { is_variable_equal = { which = dicm_bh_production_selection value = 3 } }
                set_province_flag = dicm_bh_production_probe_selected
                set_variable = { which = dicm_bh_production_selected_gain which = dicm_bh_production_workshop_gain }
            }
            else_if = {
                limit = { is_variable_equal = { which = dicm_bh_production_selection value = 4 } }
                set_province_flag = dicm_bh_production_probe_selected
                set_variable = { which = dicm_bh_production_selected_gain which = dicm_bh_production_counting_house_gain }
            }
            if = {
                limit = {
                    NOT = { trade_goods = gold }
                    NOT = { is_owned_by_trade_company = yes }
                    has_province_flag = dicm_bh_production_probe_input
                    has_province_flag = dicm_bh_production_probe_selected
                    check_variable = { which = dicm_bh_production_selected_gain value = 0.001 }
                    check_variable = { which = dicm_bh_production_selected_gain which = dicm_bh_production_minimum }
                }
                set_province_flag = dicm_bh_production_probe_pass
            }
            log = "DICM BH PRODUCTION INPUT: province=[Root.GetName] annual=[Root.dicm_bh_production_annual_value.GetValue] base=[Root.dicm_bh_production_base.GetValue] autonomy=[Root.dicm_bh_production_autonomy.GetValue] old_bonus=[Root.dicm_bh_production_old_bonus.GetValue]"
            log = "DICM BH PRODUCTION GAIN: workshop=[Root.dicm_bh_production_workshop_gain.GetValue] counting_house=[Root.dicm_bh_production_counting_house_gain.GetValue] minimum=[Root.dicm_bh_production_minimum.GetValue]"
        }
    }
    option = {
        name = dicm_bh_production_probe_edit
        province_event = { id = dicm_bh_production_probe.3 }
    }
    option = { name = dicm_bh_production_probe_close }
}
