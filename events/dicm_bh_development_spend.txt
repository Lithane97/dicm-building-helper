# Verified single-click reproduction workers and read-only diagnostics.
# Reserve-aware batches dispatch these workers from dicm_bh_development_batch events.
namespace = dicm_bh_development_spend
country_event = {
    id = dicm_bh_development_spend.1
    hidden = yes
    is_triggered_only = yes
    immediate = {
        hidden_effect = {
            dicm_bh_dev_initialize_menu = yes
            dicm_bh_save_trade_report = yes
            clr_country_flag = dicm_bh_trade_cost_mismatch
            set_variable = { which = dicm_bh_sp_status value = 0 }
            set_variable = { which = dicm_bh_sp_spent value = 0 }
            set_variable = { which = dicm_bh_sp_dev_delta value = 0 }
            set_variable = { which = dicm_bh_sp_required which = dicm_bh_dm_adm_reserve }
            dicm_bh_sp_collect_target = { TYPE = adm }
            if = {
                limit = { has_country_flag = dicm_bh_sp_target_set }
                set_country_flag = dicm_bh_sp_active
                event_target:dicm_bh_sp_target = {
                    province_event = { id = dicm_bh_development_spend.11 }
                }
                clr_country_flag = dicm_bh_sp_active
                log = "DICM BH SCRIPTED DEVELOPMENT TARGET: [dicm_bh_sp_target.GetName]"
            }
            dicm_bh_restore_trade_report = yes
            log = "DICM BH SCRIPTED DEVELOPMENT TYPE: ADM"
            log = "DICM BH SCRIPTED DEVELOPMENT STATUS: [Root.dicm_bh_sp_status.GetValue]"
            log = "DICM BH SCRIPTED DEVELOPMENT SPENT: [Root.dicm_bh_sp_spent.GetValue]"
            log = "DICM BH SCRIPTED DEVELOPMENT GAIN: [Root.dicm_bh_sp_dev_delta.GetValue]"
            if = {
                limit = { NOT = { has_country_flag = dicm_bh_db_running } }
                country_event = { id = dicm_bh_dev_menu.1 }
            }
        }
    }
    option = { name = dicm_bh_trade_close }
}
province_event = {
    id = dicm_bh_development_spend.11
    hidden = yes
    is_triggered_only = yes
    immediate = {
        hidden_effect = {
            if = {
                limit = {
                    owned_by = FROM
                    dicm_bh_sp_site_supported = yes
                    has_province_flag = dicm_bh_dev_adm_candidate
                    owner = { has_country_flag = dicm_bh_sp_active }
                }
                set_variable = { which = dicm_bh_sp_cost which = dicm_bh_dev_cost }
                export_to_variable = { which = dicm_bh_sp_dev_before value = development }
                owner = {
                    export_to_variable = { which = dicm_bh_sp_before value = trigger_value:adm_power }
                    set_variable = { which = dicm_bh_sp_cost which = PREV }
                    set_variable = { which = dicm_bh_sp_required which = dicm_bh_dm_adm_reserve }
                    change_variable = { which = dicm_bh_sp_required which = dicm_bh_sp_cost }
                    set_variable = { which = dicm_bh_sp_native_hooks_before which = num_of_times_developed }
                    set_variable = { which = dicm_bh_sp_cost_left which = dicm_bh_sp_cost }
                    if = {
                        limit = {
                            check_variable = { which = dicm_bh_sp_before which = dicm_bh_sp_required }
                            check_variable = { which = dicm_bh_sp_cost value = 1 }
                            NOT = { check_variable = { which = dicm_bh_sp_cost value = 2048 } }
                        }
                        dicm_bh_sp_power_piece = { TYPE = adm VALUE = 1024 DELTA = -1024 }
                        dicm_bh_sp_power_piece = { TYPE = adm VALUE = 512 DELTA = -512 }
                        dicm_bh_sp_power_piece = { TYPE = adm VALUE = 256 DELTA = -256 }
                        dicm_bh_sp_power_piece = { TYPE = adm VALUE = 128 DELTA = -128 }
                        dicm_bh_sp_power_piece = { TYPE = adm VALUE = 64 DELTA = -64 }
                        dicm_bh_sp_power_piece = { TYPE = adm VALUE = 32 DELTA = -32 }
                        dicm_bh_sp_power_piece = { TYPE = adm VALUE = 16 DELTA = -16 }
                        dicm_bh_sp_power_piece = { TYPE = adm VALUE = 8 DELTA = -8 }
                        dicm_bh_sp_power_piece = { TYPE = adm VALUE = 4 DELTA = -4 }
                        dicm_bh_sp_power_piece = { TYPE = adm VALUE = 2 DELTA = -2 }
                        dicm_bh_sp_power_piece = { TYPE = adm VALUE = 1 DELTA = -1 }
                        export_to_variable = { which = dicm_bh_sp_after value = trigger_value:adm_power }
                        set_variable = { which = dicm_bh_sp_spent which = dicm_bh_sp_before }
                        subtract_variable = { which = dicm_bh_sp_spent which = dicm_bh_sp_after }
                        if = {
                            limit = {
                                is_variable_equal = { which = dicm_bh_sp_spent which = dicm_bh_sp_cost }
                                check_variable = { which = dicm_bh_sp_after which = dicm_bh_dm_adm_reserve }
                            }
                            set_variable = { which = dicm_bh_sp_status value = 1 }
                        }
                        else = {
                            # Failed payment verification: refund observed deduction, no development.
                            set_variable = { which = dicm_bh_sp_cost_left which = dicm_bh_sp_spent }
                            dicm_bh_sp_power_piece = { TYPE = adm VALUE = 1024 DELTA = 1024 }
                            dicm_bh_sp_power_piece = { TYPE = adm VALUE = 512 DELTA = 512 }
                            dicm_bh_sp_power_piece = { TYPE = adm VALUE = 256 DELTA = 256 }
                            dicm_bh_sp_power_piece = { TYPE = adm VALUE = 128 DELTA = 128 }
                            dicm_bh_sp_power_piece = { TYPE = adm VALUE = 64 DELTA = 64 }
                            dicm_bh_sp_power_piece = { TYPE = adm VALUE = 32 DELTA = 32 }
                            dicm_bh_sp_power_piece = { TYPE = adm VALUE = 16 DELTA = 16 }
                            dicm_bh_sp_power_piece = { TYPE = adm VALUE = 8 DELTA = 8 }
                            dicm_bh_sp_power_piece = { TYPE = adm VALUE = 4 DELTA = 4 }
                            dicm_bh_sp_power_piece = { TYPE = adm VALUE = 2 DELTA = 2 }
                            dicm_bh_sp_power_piece = { TYPE = adm VALUE = 1 DELTA = 1 }
                            set_variable = { which = dicm_bh_sp_status value = 2 }
                        }
                    }
                }
                if = {
                    limit = { owner = { is_variable_equal = { which = dicm_bh_sp_status value = 1 } } }
                    add_base_tax = 1
                    export_to_variable = { which = dicm_bh_sp_dev_after value = development }
                    set_variable = { which = dicm_bh_sp_raw_delta which = dicm_bh_sp_dev_after }
                    subtract_variable = { which = dicm_bh_sp_raw_delta which = dicm_bh_sp_dev_before }
                    owner = {
                        set_variable = { which = dicm_bh_sp_dev_before which = PREV }
                        set_variable = { which = dicm_bh_sp_raw_delta which = PREV }
                        set_variable = { which = dicm_bh_sp_auto_hooks_delta which = num_of_times_developed }
                        subtract_variable = { which = dicm_bh_sp_auto_hooks_delta which = dicm_bh_sp_native_hooks_before }
                    }
                    if = {
                        limit = {
                            is_variable_equal = { which = dicm_bh_sp_raw_delta value = 1 }
                            owner = { is_variable_equal = { which = dicm_bh_sp_auto_hooks_delta value = 0 } }
                        }
                        # Verified +1 before any institution mutation; helper restores it.
                        dicm_bh_sp_apply_institution_progress = { BASE = tax }
                        if = {
                            limit = { owner = { is_variable_equal = { which = dicm_bh_sp_status value = 1 } } }
                            dicm_bh_sp_apply_side_effects = yes
                            set_province_flag = dicm_bh_sp_adm_replay_pending
                            province_event = { id = dicm_bh_development_spend.21 }
                        }
                    }
                    else_if = {
                        limit = { is_variable_equal = { which = dicm_bh_sp_raw_delta value = 0 } }
                        owner = {
                            dicm_bh_sp_refund_payment = { TYPE = adm }
                            set_variable = { which = dicm_bh_sp_status value = 4 }
                        }
                    }
                    else = {
                        # Unexpected base delta or automatic hooks: do not double-replay.
                        owner = { set_variable = { which = dicm_bh_sp_status value = 3 } }
                    }
                    export_to_variable = { which = dicm_bh_sp_dev_delta value = development }
                    subtract_variable = { which = dicm_bh_sp_dev_delta which = dicm_bh_sp_dev_before }
                    owner = { set_variable = { which = dicm_bh_sp_dev_delta which = PREV } }
                }
            }
        }
    }
    option = { name = dicm_bh_trade_close }
}
