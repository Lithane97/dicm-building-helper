# Console diagnostics only. No construction or spending.
# event dicm_bh_tax_probe.1            -> capital province
# event dicm_bh_tax_probe.2 PROVINCE_ID -> specified province
namespace = dicm_bh_tax_probe
country_event = {
    id = dicm_bh_tax_probe.1
    hidden = yes
    is_triggered_only = yes
    immediate = {
        capital_scope = { province_event = { id = dicm_bh_tax_probe.2 } }
    }
    option = { name = dicm_bh_tax_probe_close }
}
province_event = {
    id = dicm_bh_tax_probe.2
    title = dicm_bh_tax_probe_title
    desc = dicm_bh_tax_probe_desc
    picture = TRADE_eventPicture
    is_triggered_only = yes
    immediate = {
        dicm_bh_calculate_tax_gains = yes
        set_variable = { which = dicm_bh_tax_minimum value = 0.1 }
        set_variable = { which = dicm_bh_tax_selection value = 0 }
        set_variable = { which = dicm_bh_tax_selected_gain value = 0 }
        clr_province_flag = dicm_bh_tax_probe_selected
        clr_province_flag = dicm_bh_tax_probe_pass
        owner = {
            set_variable = { which = dicm_bh_tax_minimum which = dicm_bh_minimum }
            set_variable = { which = dicm_bh_tax_selection which = dicm_bh_selection }
            ROOT = {
                set_variable = { which = dicm_bh_tax_minimum which = PREV }
                set_variable = { which = dicm_bh_tax_selection which = PREV }
            }
        }
        if = {
            limit = { is_variable_equal = { which = dicm_bh_tax_selection value = 1 } }
            set_province_flag = dicm_bh_tax_probe_selected
            set_variable = { which = dicm_bh_tax_selected_gain which = dicm_bh_tax_church_gain }
        }
        else_if = {
            limit = { is_variable_equal = { which = dicm_bh_tax_selection value = 2 } }
            set_province_flag = dicm_bh_tax_probe_selected
            set_variable = { which = dicm_bh_tax_selected_gain which = dicm_bh_tax_cathedral_gain }
        }
        if = {
            limit = {
                has_province_flag = dicm_bh_tax_probe_selected
                check_variable = { which = dicm_bh_tax_selected_gain value = 0.001 }
                check_variable = { which = dicm_bh_tax_selected_gain which = dicm_bh_tax_minimum }
            }
            set_province_flag = dicm_bh_tax_probe_pass
        }
        log = "DICM BH TAX PROBE: province=[Root.GetName] base=[Root.dicm_bh_tax_base.GetValue] flat_annual=[Root.dicm_bh_tax_flat.GetValue] autonomy=[Root.dicm_bh_tax_autonomy.GetValue] local_tax_modifier=[Root.dicm_bh_tax_local_modifier.GetValue] existing_bonus=[Root.dicm_bh_tax_old_bonus.GetValue] church_monthly=[Root.dicm_bh_tax_church_gain.GetValue] cathedral_monthly=[Root.dicm_bh_tax_cathedral_gain.GetValue] minimum=[Root.dicm_bh_tax_minimum.GetValue]"
    }
    option = { name = dicm_bh_tax_probe_close }
}
