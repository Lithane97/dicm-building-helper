# Read-only diagnostic. Estimates are NOT yet validated against the native macrobuilder.
namespace = dicm_bh_probe
country_event = {
    id = dicm_bh_probe.1
    title = dicm_bh_probe_title
    desc = dicm_bh_probe_desc
    picture = TRADE_eventPicture
    is_triggered_only = yes
    immediate = {
        clr_country_flag = dicm_bh_probe_supported
        clr_country_flag = dicm_bh_probe_pass
        clr_country_flag = dicm_bh_probe_percent_defined
        set_variable = { which = dicm_bh_probe_percent value = 0 }
        dicm_bh_load_selected_scale = yes
        set_variable = { which = dicm_bh_probe_target value = 0 }
        set_variable = { which = dicm_bh_probe_gain_local value = 0 }
        set_variable = { which = dicm_bh_probe_gain_combined value = 0 }
        if = {
            limit = { is_variable_equal = { which = dicm_bh_selection value = 5 } }
            set_variable = { which = dicm_bh_probe_target value = 0.5 }
            set_country_flag = dicm_bh_probe_supported
        }
        else_if = {
            limit = { is_variable_equal = { which = dicm_bh_selection value = 6 } }
            set_variable = { which = dicm_bh_probe_target value = 1 }
            set_country_flag = dicm_bh_probe_supported
        }
        else_if = {
            limit = { is_variable_equal = { which = dicm_bh_selection value = 7 } }
            set_variable = { which = dicm_bh_probe_target value = 1.25 }
            set_country_flag = dicm_bh_probe_supported
        }
        capital_scope = {
            export_to_variable = { which = dicm_bh_probe_tp value = province_trade_power }
            export_to_variable = { which = dicm_bh_probe_local value = modifier:province_trade_power_modifier }
            export_to_variable = { which = dicm_bh_probe_flat value = modifier:province_trade_power_value }
            export_to_variable = { which = dicm_bh_probe_dev value = development }
            export_to_variable = { which = dicm_bh_probe_autonomy value = local_autonomy }
            set_variable = { which = dicm_bh_probe_old value = 0 }
            if = {
                limit = { has_building = stock_exchange }
                set_variable = { which = dicm_bh_probe_old value = 1.25 }
            }
            else_if = {
                limit = { has_building = trade_depot }
                set_variable = { which = dicm_bh_probe_old value = 1 }
            }
            else_if = {
                limit = { has_building = marketplace }
                set_variable = { which = dicm_bh_probe_old value = 0.5 }
            }
            # Copy exported inputs to the country for the diagnostic event text.
            ROOT = {
                set_variable = { which = dicm_bh_probe_tp which = PREV }
                set_variable = { which = dicm_bh_probe_local which = PREV }
                set_variable = { which = dicm_bh_probe_flat which = PREV }
                set_variable = { which = dicm_bh_probe_dev which = PREV }
                set_variable = { which = dicm_bh_probe_autonomy which = PREV }
                set_variable = { which = dicm_bh_probe_old which = PREV }
            }
        }
        export_to_variable = { which = dicm_bh_probe_global value = modifier:global_prov_trade_power_modifier }
        set_variable = { which = dicm_bh_probe_delta which = dicm_bh_probe_target }
        subtract_variable = { which = dicm_bh_probe_delta which = dicm_bh_probe_old }
        if = {
            limit = { has_country_flag = dicm_bh_probe_supported check_variable = { which = dicm_bh_probe_delta value = 0.001 } }
            # Candidate A: undo the local additive multiplier, preserving other factors.
            set_variable = { which = dicm_bh_probe_den_local which = dicm_bh_probe_local }
            change_variable = { which = dicm_bh_probe_den_local value = 1 }
            if = {
                limit = { check_variable = { which = dicm_bh_probe_den_local value = 0.001 } }
                set_variable = { which = dicm_bh_probe_gain_local which = dicm_bh_probe_tp }
                divide_variable = { which = dicm_bh_probe_gain_local which = dicm_bh_probe_den_local }
                multiply_variable = { which = dicm_bh_probe_gain_local which = dicm_bh_probe_delta }
            }
            # Candidate B tests whether the national provincial modifier must be included.
            set_variable = { which = dicm_bh_probe_den_combined which = dicm_bh_probe_den_local }
            change_variable = { which = dicm_bh_probe_den_combined which = dicm_bh_probe_global }
            if = {
                limit = { check_variable = { which = dicm_bh_probe_den_combined value = 0.001 } }
                set_variable = { which = dicm_bh_probe_gain_combined which = dicm_bh_probe_tp }
                divide_variable = { which = dicm_bh_probe_gain_combined which = dicm_bh_probe_den_combined }
                multiply_variable = { which = dicm_bh_probe_gain_combined which = dicm_bh_probe_delta }
            }
            # Relative gain = additional trade power / current trade power * 100.
            # A zero baseline has no finite percentage; report it rather than divide by zero.
            if = {
                limit = { check_variable = { which = dicm_bh_probe_tp value = 0.001 } }
                set_country_flag = dicm_bh_probe_percent_defined
                set_variable = { which = dicm_bh_probe_percent which = dicm_bh_probe_gain_combined }
                multiply_variable = { which = dicm_bh_probe_percent value = 100 }
                divide_variable = { which = dicm_bh_probe_percent which = dicm_bh_probe_tp }
                if = {
                    limit = { check_variable = { which = dicm_bh_probe_gain_combined which = dicm_bh_minimum } }
                    set_country_flag = dicm_bh_probe_pass
                }
            }
        }
        log = "DICM BH PROBE: selected=[Root.GetDICMBHBuilding] minimum_absolute=[Root.dicm_bh_minimum.GetValue] trade_power=[Root.dicm_bh_probe_tp.GetValue] local_modifier=[Root.dicm_bh_probe_local.GetValue] global_modifier=[Root.dicm_bh_probe_global.GetValue] target_bonus=[Root.dicm_bh_probe_target.GetValue] existing_bonus=[Root.dicm_bh_probe_old.GetValue] delta=[Root.dicm_bh_probe_delta.GetValue] candidate_A=[Root.dicm_bh_probe_gain_local.GetValue] candidate_B=[Root.dicm_bh_probe_gain_combined.GetValue] gain_percent=[Root.dicm_bh_probe_percent.GetValue]"
    }
    option = { name = dicm_bh_probe_close }
}
