# Province scope, country ROOT. Additional national force-limit capacity.
# Flat building contribution is reduced by autonomy, then multiplied by the
# owner's national force-limit modifier. Other additive flat sources cancel.
# No sampling, resource changes, buildings or completion effects.
dicm_bh_calculate_force_gain = {
    export_to_variable = { which = dicm_bh_force_autonomy value = local_autonomy }
    set_variable = { which = dicm_bh_force_autonomy_factor which = dicm_bh_force_autonomy }
    divide_variable = { which = dicm_bh_force_autonomy_factor value = 100 }
    multiply_variable = { which = dicm_bh_force_autonomy_factor value = -1 }
    change_variable = { which = dicm_bh_force_autonomy_factor value = 1 }
    if = {
        limit = { NOT = { check_variable = { which = dicm_bh_force_autonomy_factor value = 0 } } }
        set_variable = { which = dicm_bh_force_autonomy_factor value = 0 }
    }
    if = {
        limit = { check_variable = { which = dicm_bh_force_autonomy_factor value = 1 } }
        set_variable = { which = dicm_bh_force_autonomy_factor value = 1 }
    }
    owner = {
        export_to_variable = { which = dicm_bh_force_country_modifier value = modifier:$TYPE$_forcelimit_modifier }
        PREV = { set_variable = { which = dicm_bh_force_country_modifier which = PREV } }
    }
    set_variable = { which = dicm_bh_force_country_factor which = dicm_bh_force_country_modifier }
    change_variable = { which = dicm_bh_force_country_factor value = 1 }
    if = {
        limit = { NOT = { check_variable = { which = dicm_bh_force_country_factor value = 0 } } }
        set_variable = { which = dicm_bh_force_country_factor value = 0 }
    }
    set_variable = { which = dicm_bh_force_old_bonus value = 0 }
    if = {
        limit = { has_building = $UPPER$ }
        set_variable = { which = dicm_bh_force_old_bonus value = $UPPER_BONUS$ }
    }
    else_if = {
        limit = { has_building = $LOWER$ }
        set_variable = { which = dicm_bh_force_old_bonus value = $LOWER_BONUS$ }
    }
    set_variable = { which = dicm_bh_force_delta value = $BONUS$ }
    subtract_variable = { which = dicm_bh_force_delta which = dicm_bh_force_old_bonus }
    if = {
        limit = { NOT = { check_variable = { which = dicm_bh_force_delta value = 0 } } }
        set_variable = { which = dicm_bh_force_delta value = 0 }
    }
    set_variable = { which = dicm_bh_force_province_gain which = dicm_bh_force_delta }
    multiply_variable = { which = dicm_bh_force_province_gain which = dicm_bh_force_autonomy_factor }
    set_variable = { which = dicm_bh_trade_gain which = dicm_bh_force_province_gain }
    multiply_variable = { which = dicm_bh_trade_gain which = dicm_bh_force_country_factor }
}
