    dicm_bh_save_minimum = yes
    set_variable = { which = dicm_bh_scale value = $index$ }
    if = {
        limit = { NOT = { has_country_flag = dicm_bh_initialized_$category$ } }
        set_variable = { which = dicm_bh_saved_$category$ value = $default$ }
        set_country_flag = dicm_bh_initialized_$category$

    set_variable = { which = dicm_bh_minimum which = dicm_bh_saved_$category$ }
    set_variable = { which = dicm_bh_step_small value = $small$ }
    set_variable = { which = dicm_bh_step_large value = $large$ }
}

            NOT = {
