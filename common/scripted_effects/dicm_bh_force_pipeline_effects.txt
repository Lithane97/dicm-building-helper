# Country ROOT. Both preview and live construction share the same collection.
# LOWER/UPPER identify the army or navy force-limit building family.
dicm_bh_collect_force_candidates = {
    every_owned_province = {
        limit = {
            dicm_bh_trade_province_free = yes
            can_build = $BUILDING$
            NOT = { has_building = $BUILDING$ }
            OR = {
                AND = {
                    owner = { has_country_flag = dicm_bh_trade_upgrade_mode }
                    has_building = $LOWER$
                }
                AND = {
                    owner = { NOT = { has_country_flag = dicm_bh_trade_upgrade_mode } }
                    NOT = { OR = { has_building = $LOWER$ has_building = $UPPER$ } }
                }
            }
        }
        dicm_bh_calculate_force_gain = { TYPE = $TYPE$ LOWER = $LOWER$ UPPER = $UPPER$ BONUS = $BONUS$ LOWER_BONUS = $LOWER_BONUS$ UPPER_BONUS = $UPPER_BONUS$ }
        set_variable = { which = dicm_bh_trade_old_cost value = 0 }
        if = {
            limit = { has_building = $LOWER$ }
            set_variable = { which = dicm_bh_trade_old_cost value = $LOWER_COST$ }
        }
        dicm_bh_calculate_trade_cost_factor = yes
        set_variable = { which = dicm_bh_trade_cost value = $BASE_COST$ }
        dicm_bh_apply_trade_cost_factor = yes
        dicm_bh_mark_trade_candidate = yes
    }
}
