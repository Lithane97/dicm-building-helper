    clr_country_flag = dicm_bh_trade_best_set

# Country scope. VALUE is the source variable name, supplied as a literal parameter.
# Write VALUE_fmt_* groups used by comma-separated money localisation.
# Preserve the source amount; rounding and zero padding are for display only.
# Example: dicm_bh_format_money = { VALUE = dicm_bh_est_build_cost }
dicm_bh_format_money = {
    set_variable = { which = $VALUE$_fmt_whole which = $VALUE$ }
    round_variable = { which = $VALUE$_fmt_whole value = -1 }
    set_variable = { which = $VALUE$_fmt_cents which = $VALUE$ }
    subtract_variable = { which = $VALUE$_fmt_cents which = $VALUE$_fmt_whole }
    multiply_variable = { which = $VALUE$_fmt_cents value = 100 }
    round_variable = { which = $VALUE$_fmt_cents value = 0 }
    if = {
        limit = { check_variable = { which = $VALUE$_fmt_cents value = 100 } }
        change_variable = { which = $VALUE$_fmt_whole value = 1 }
        set_variable = { which = $VALUE$_fmt_cents value = 0 }
    }
    set_variable = { which = $VALUE$_fmt_thousands which = $VALUE$_fmt_whole }
    divide_variable = { which = $VALUE$_fmt_thousands value = 1000 }
    round_variable = { which = $VALUE$_fmt_thousands value = -1 }
    set_variable = { which = $VALUE$_fmt_ones which = $VALUE$_fmt_thousands }
    multiply_variable = { which = $VALUE$_fmt_ones value = 1000 }
    subtract_variable = { which = $VALUE$_fmt_ones which = $VALUE$_fmt_whole }
    multiply_variable = { which = $VALUE$_fmt_ones value = -1 }
    set_variable = { which = $VALUE$_fmt_millions which = $VALUE$_fmt_thousands }
    divide_variable = { which = $VALUE$_fmt_millions value = 1000 }
    round_variable = { which = $VALUE$_fmt_millions value = -1 }
    set_variable = { which = $VALUE$_fmt_middle which = $VALUE$_fmt_millions }
    multiply_variable = { which = $VALUE$_fmt_middle value = 1000 }
    subtract_variable = { which = $VALUE$_fmt_middle which = $VALUE$_fmt_thousands }
    multiply_variable = { which = $VALUE$_fmt_middle value = -1 }
}
