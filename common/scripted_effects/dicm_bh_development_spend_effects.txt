# Single-click selection, payment support and reproduced side effects.
# Reserve-aware batches invoke the same native event workers; no scheduler.
# Selection/evaluation: country ROOT. Replay worker: province ROOT, country FROM.
dicm_bh_sp_collect_target = {
    clr_country_flag = dicm_bh_sp_target_set
    dicm_bh_evaluate_country_development = yes
    export_to_variable = { which = dicm_bh_sp_power value = trigger_value:$TYPE$_power }
    every_owned_province = {
        limit = {
            dicm_bh_sp_site_supported = yes
            has_province_flag = dicm_bh_dev_$TYPE$_candidate
            check_variable = { which = dicm_bh_dev_cost value = 1 }
            NOT = { check_variable = { which = dicm_bh_dev_cost value = 2048 } }
            owner = { NOT = { has_country_flag = dicm_bh_trade_cost_mismatch } }
        }
        set_variable = { which = dicm_bh_sp_required which = ROOT }
        # Country required slot is initialized with the chosen reserve by caller.
        change_variable = { which = dicm_bh_sp_required which = dicm_bh_dev_cost }
        set_variable = { which = dicm_bh_sp_power which = ROOT }
        if = {
            limit = { check_variable = { which = dicm_bh_sp_power which = dicm_bh_sp_required } }
            if = {
                limit = { owner = { NOT = { has_country_flag = dicm_bh_sp_target_set } } }
                save_event_target_as = dicm_bh_sp_target
                owner = { set_country_flag = dicm_bh_sp_target_set }
            }
            else_if = {
                limit = {
                    check_variable = { which = dicm_bh_dev_$TYPE$_score which = event_target:dicm_bh_sp_target }
                    OR = {
                        NOT = { is_variable_equal = { which = dicm_bh_dev_$TYPE$_score which = event_target:dicm_bh_sp_target } }
                        NOT = { check_variable = { which = dicm_bh_dev_cost which = event_target:dicm_bh_sp_target } }
                    }
                }
                save_event_target_as = dicm_bh_sp_target
            }
        }
    }
}

# Country scope. Literal denominations avoid assuming variable-valued power effects.
dicm_bh_sp_power_piece = {
    if = {
        limit = { check_variable = { which = dicm_bh_sp_cost_left value = $VALUE$ } }
        add_$TYPE$_power = $DELTA$
        subtract_variable = { which = dicm_bh_sp_cost_left value = $VALUE$ }
    }
}

# Province scope. Native add_devastation accepts whole numbers only.
# User-approved approximation: nearest whole point, half values rounded upward.
# Apply integer denominations only; fractional native effects silently truncate.
dicm_bh_sp_devastation_piece = {
    if = {
        limit = { check_variable = { which = dicm_bh_sp_devastation_units value = $UNITS$ } }
        add_devastation = $DELTA$
        subtract_variable = { which = dicm_bh_sp_devastation_units value = $UNITS$ }
    }
}

# Province scope, only after a verified paid +1. Raw base effects do not replay
# development hooks (validated separately). The scaled institution effect includes
# the next point itself, so briefly expose pre-click development, then restore it.
# Keeping this after verification prevents institution progress on a failed +1.
dicm_bh_sp_apply_institution_progress = {
    add_base_$BASE$ = -1
    export_to_variable = { which = dicm_bh_sp_institution_sample_dev value = development }
    if = {
        limit = { is_variable_equal = { which = dicm_bh_sp_institution_sample_dev which = dicm_bh_sp_dev_before } }
        add_next_institution_embracement_scaled = 5
        add_base_$BASE$ = 1
    }
    else = { owner = { set_variable = { which = dicm_bh_sp_status value = 3 } } }
    export_to_variable = { which = dicm_bh_sp_institution_sample_dev value = development }
    if = {
        limit = { NOT = { is_variable_equal = { which = dicm_bh_sp_institution_sample_dev which = dicm_bh_sp_dev_after } } }
        owner = { set_variable = { which = dicm_bh_sp_status value = 3 } }
    }
}

dicm_bh_sp_apply_side_effects = {
    # Estimated reduction: resulting development * native scale / 6.
    # Round ONCE before integer effect dispatch; do not truncate each fraction.
    set_variable = { which = dicm_bh_sp_devastation_units which = dicm_bh_sp_dev_after }
    multiply_variable = { which = dicm_bh_sp_devastation_units value = 5 }
    divide_variable = { which = dicm_bh_sp_devastation_units value = 6 }
    if = {
        limit = { check_variable = { which = dicm_bh_sp_devastation_units value = 100 } }
        set_variable = { which = dicm_bh_sp_devastation_units value = 100 }
    }
    change_variable = { which = dicm_bh_sp_devastation_units value = 0.5 }
    round_variable = { which = dicm_bh_sp_devastation_units value = -1 }
    set_variable = { which = dicm_bh_sp_devastation_rounded which = dicm_bh_sp_devastation_units }
    dicm_bh_sp_devastation_piece = { UNITS = 64 DELTA = -64 }
    dicm_bh_sp_devastation_piece = { UNITS = 32 DELTA = -32 }
    dicm_bh_sp_devastation_piece = { UNITS = 16 DELTA = -16 }
    dicm_bh_sp_devastation_piece = { UNITS = 8 DELTA = -8 }
    dicm_bh_sp_devastation_piece = { UNITS = 4 DELTA = -4 }
    dicm_bh_sp_devastation_piece = { UNITS = 2 DELTA = -2 }
    dicm_bh_sp_devastation_piece = { UNITS = 1 DELTA = -1 }
    # Snapshot every estate's whole-number influence, then deduct weighted land.
    owner = { dicm_bh_sp_apply_estate_land = yes }
}


# Country scope; refund only observed payment after a failed base increase.
dicm_bh_sp_refund_payment = {
    set_variable = { which = dicm_bh_sp_cost_left which = dicm_bh_sp_spent }
    dicm_bh_sp_power_piece = { TYPE = $TYPE$ VALUE = 1024 DELTA = 1024 }
    dicm_bh_sp_power_piece = { TYPE = $TYPE$ VALUE = 512 DELTA = 512 }
    dicm_bh_sp_power_piece = { TYPE = $TYPE$ VALUE = 256 DELTA = 256 }
    dicm_bh_sp_power_piece = { TYPE = $TYPE$ VALUE = 128 DELTA = 128 }
    dicm_bh_sp_power_piece = { TYPE = $TYPE$ VALUE = 64 DELTA = 64 }
    dicm_bh_sp_power_piece = { TYPE = $TYPE$ VALUE = 32 DELTA = 32 }
    dicm_bh_sp_power_piece = { TYPE = $TYPE$ VALUE = 16 DELTA = 16 }
    dicm_bh_sp_power_piece = { TYPE = $TYPE$ VALUE = 8 DELTA = 8 }
    dicm_bh_sp_power_piece = { TYPE = $TYPE$ VALUE = 4 DELTA = 4 }
    dicm_bh_sp_power_piece = { TYPE = $TYPE$ VALUE = 2 DELTA = 2 }
    dicm_bh_sp_power_piece = { TYPE = $TYPE$ VALUE = 1 DELTA = 1 }
    export_to_variable = { which = dicm_bh_sp_after value = trigger_value:$TYPE$_power }
    set_variable = { which = dicm_bh_sp_spent which = dicm_bh_sp_before }
    subtract_variable = { which = dicm_bh_sp_spent which = dicm_bh_sp_after }
}
